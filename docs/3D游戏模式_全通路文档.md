## 一、架构概览

### 1.1 系统架构

```
┌──────────────────────┐     Socket.IO      ┌──────────────────────┐
│   Unity 3D Client    │ ◄────10Hz──────► │   Node.js Server      │
│                      │   position sync    │                      │
│  GameManager3D.cs    │ ◄───events──────► │  Game.js (3D mode)   │
│  PlayerController3D  │   game:state       │  PositionSync.js     │
│  NetworkManager      │   night:action     │  submit3DAttack()    │
│  VillageSceneSetup   │   3d:attack/hide   │  submit3DHide()      │
└──────────────────────┘                    └──────────────────────┘
         │                                          │
         │ 同一房间内的其他客户端                      │
         │ (React Web / 其他 Unity 实例)              │
         ▼                                          ▼
┌──────────────────────┐                    ┌──────────────────────┐
│  React Web Client    │                    │  Other Unity Client  │
│  (观看/桌游模式)      │                    │  (其他3D玩家)        │
└──────────────────────┘                    └──────────────────────┘
```

### 1.2 数据流

```
房主创建房间 (gameMode: 'THIRD_PERSON')
  → 玩家加入
  → 开始游戏
  → 选人阶段 (跳过，随机分配)
  → 序幕 (8秒)
  → 夜晚: 3D自由行动 (120秒)
    → Unity: 玩家WASD移动、蚀者攻击、灵织者藏匿
    → Server: PositionSync 10Hz广播位置
    → Server: 攻击判定 (8秒冷却、30%基础逃脱率)
  → 夜晚结束: NightResolver 标准结算
  → 白天: 投票放逐 (与桌游模式相同)
  → 循环直到游戏结束
```

```csharp
public class VillageSceneSetup : MonoBehaviour
{
    [Header("Village Layout")]
    public int houseCount = 12;
    public float villageRadius = 30f;
    public GameObject housePrefab;
    public GameObject[] landmarkPrefabs; // [0]=Tree, [1]=Well, [2]=Blacksmith, [3]=Graveyard

    [Header("Lighting")]
    public Light moonLight;    // 夜晚定向光 (冷色调, ~0.3 intensity)
    public Light sunLight;     // 白天定向光 (暖色调, ~1.0 intensity)
    public Material nightSkybox;
    public Material daySkybox;

    void Awake()
    {
        GenerateVillage();
    }

    void GenerateVillage()
    {
        for (int i = 0; i < houseCount; i++)
        {
            float angle = (360f / houseCount) * i;
            Vector3 pos = new Vector3(
                Mathf.Sin(angle * Mathf.Deg2Rad) * villageRadius,
                0,
                Mathf.Cos(angle * Mathf.Deg2Rad) * villageRadius
            );
            // 房屋面向圆心
            Quaternion rot = Quaternion.LookRotation(-pos.normalized);
            GameObject house = Instantiate(housePrefab, pos, rot);
            house.name = GetHouseName(i);
            housePositions[i] = pos;
        }
    }

    string GetHouseName(int index)
    {
        // 房屋名称映射到游戏角色主题
        string[] names = {
            "追猎者之屋", "帷幕学者之塔", "灵愈师之居",
            "草药贤者之庐", "灵织者之舍", "帷幕守卫之府",
            "冥僧人之院", "蚀者之穴", "老兵之宅",
            "行商之店", "学徒之室", "史官之阁"
        };
        return names[index % names.Length];
    }
}
```

```csharp
public class GameManager3D : MonoBehaviour
{
    public Light moonLight;
    public Light sunLight;
    public float transitionDuration = 3f;

    // 由服务端 phaseChange 事件触发
    public void OnPhaseChanged(string phase)
    {
        switch (phase)
        {
            case "NIGHT":
                StartCoroutine(TransitionToNight());
                break;
            case "DAY":
                StartCoroutine(TransitionToDay());
                break;
        }
    }

    IEnumerator TransitionToNight()
    {
        float elapsed = 0;
        while (elapsed < transitionDuration)
        {
            elapsed += Time.deltaTime;
            float t = elapsed / transitionDuration;
            sunLight.intensity = Mathf.Lerp(1.0f, 0f, t);
            moonLight.intensity = Mathf.Lerp(0f, 0.3f, t);
            RenderSettings.ambientIntensity = Mathf.Lerp(1.0f, 0.3f, t);
            yield return null;
        }
    }
}
```

```python
import bpy
import os
import sys

# 获取命令行参数
output_dir = sys.argv[-1] if sys.argv[-1].endswith('/') else "./output/"

characters = [
    "sigurd", "freyja", "morrigan", "kakhet", "hector",
    "romulus", "skadi", "halvard", "brigid", "voidboat"
]

for char in characters:
    script_path = f"characters/{char}.py"
    if os.path.exists(script_path):
        exec(open(script_path).read())
        
        # 导出 FBX (带绑定)
        bpy.ops.export_scene.fbx(
            filepath=f"{output_dir}/{char}.fbx",
            use_selection=True,
            bake_space_transform=True,
            mesh_smooth_type='FACE'
        )
        
        # 导出 GLB (Web查看)
        bpy.ops.export_scene.gltf(
            filepath=f"{output_dir}/{char}.glb",
            use_selection=True,
            export_format='GLB'
        )
        
        print(f"✓ {char} 导出完成")
        bpy.ops.object.select_all(action='SELECT')
        bpy.ops.object.delete()
```

```csharp
static void BatchImportCharacters()
{
    string importPath = "Assets/Models/Characters";
    string[] fbxFiles = Directory.GetFiles(importPath, "*.fbx", SearchOption.AllDirectories);

    foreach (string fbx in fbxFiles)
    {
        string relativePath = "Assets" + fbx.Substring(Application.dataPath.Length);
        ModelImporter importer = AssetImporter.GetAtPath(relativePath) as ModelImporter;

        importer.useFileScale = false;
        importer.meshCompression = ModelImporterMeshCompression.Medium;
```

```csharp
using System;
using System.Collections;
using System.Collections.Generic;
using UnityEngine;
using SocketIOClient;
using SocketIOClient.Newtonsoft.Json;
using Newtonsoft.Json;
using UnityEngine.Events;

[System.Serializable]
public class PlayerState
{
    public string id;
    public string name;
    public string characterId;
    public string role;
    public bool alive;
    public bool isBot;
    public Vector3 position;
    public Quaternion rotation;
    public bool isHidden;
    public bool isSprinting;
}

[System.Serializable]
public class GameState
{
    public string id;
    public string phase;
    public int round;
    public string nightStep;
    public int timeLeft;
    public string gameMode;
    public List<PlayerState> players;
    public Dictionary<string, string> votes;
    public NightEventLog[] nightLog;
}

[System.Serializable]
public class NightEventLog
{
    public string type;
    public string player;
    public string target;
    public string msg;
}

public class NetworkManager : MonoBehaviour
{
    public static NetworkManager Instance { get; private set; }

    [Header("Connection")]
    public string serverUrl = "http://210.16.170.144:4000";
    public string playerName = "UnityPlayer";
    public string roomCode = "";

    private SocketIOUnity socket;

    // 状态
    public GameState CurrentGameState { get; private set; }
    public PrivateState MyPrivateState { get; private set; }
    public bool IsConnected { get; private set; }

    // 事件
    public UnityEvent<GameState> OnGameStateChanged;
    public UnityEvent<string> OnPhaseChanged;
    public UnityEvent<string> OnNightStepChanged;
    public UnityEvent<Vector3, string> OnPlayerPositionUpdate;

    void Awake()
    {
        if (Instance == null)
        {
            Instance = this;
            DontDestroyOnLoad(gameObject);
        }
        else
        {
            Destroy(gameObject);
            return;
        }
    }

    void Start()
    {
        Connect();
    }

    #region Connection

    public void Connect()
    {
        var uri = new Uri(serverUrl);
        socket = new SocketIOUnity(uri, new SocketIOOptions
        {
            Transport = SocketIOClient.Transport.TransportProtocol.WebSocket,
            Reconnection = true,
            ReconnectionAttempts = 10,
            ReconnectionDelay = 1000
        });

        socket.JsonSerializer = new NewtonsoftJsonSerializer();

        RegisterEvents();

        socket.ConnectAsync();
        Debug.Log($"[Network] Connecting to {serverUrl}...");
    }

    void RegisterEvents()
    {
        socket.OnConnected += (sender, e) =>
        {
            Debug.Log("[Network] Connected! ID: " + socket.Id);
            IsConnected = true;
        };

        socket.OnDisconnected += (sender, e) =>
        {
            Debug.LogWarning($"[Network] Disconnected: {e}");
            IsConnected = false;
        };

        socket.OnError += (sender, e) =>
        {
            Debug.LogError($"[Network] Error: {e}");
        };

        // 游戏状态
        socket.On("game:state", response =>
        {
            string json = response.GetValue(0).ToString();
            CurrentGameState = JsonConvert.DeserializeObject<GameState>(json);
            OnGameStateChanged?.Invoke(CurrentGameState);
        });

        // 私有状态
        socket.On("game:privateState", response =>
        {
            string json = response.GetValue(0).ToString();
            MyPrivateState = JsonConvert.DeserializeObject<PrivateState>(json);
        });

        // 阶段切换
        socket.On("game:phaseChange", response =>
        {
            string phase = response.GetValue<JsonElement>(0).GetProperty("phase").GetString();
            OnPhaseChanged?.Invoke(phase);
        });

        // 夜晚步骤
        socket.On("game:nightStep", response =>
        {
            string step = response.GetValue<JsonElement>(0).GetProperty("nightStep").GetString();
            OnNightStepChanged?.Invoke(step);
        });

        // 位置同步 (3D模式)
        socket.On("game:positionSync", response =>
        {
            var data = response.GetValue<JsonElement>(0);
            var positions = data.GetProperty("positions");
            foreach (var pos in positions.EnumerateArray())
            {
                string playerId = pos.GetProperty("playerId").GetString();
                float x = pos.GetProperty("x").GetSingle();
                float y = pos.GetProperty("y").GetSingle();
                float z = pos.GetProperty("z").GetSingle();
                OnPlayerPositionUpdate?.Invoke(new Vector3(x, y, z), playerId);
            }
        });

        // 游戏结束
        socket.On("game:over", response =>
        {
            Debug.Log("[Network] Game Over!");
        });
    }

    #endregion

    #region Outgoing Events

    public void CreateRoom(string name, string gameMode = "THIRD_PERSON")
    {
        socket.Emit("room:create", new
        {
            playerName = name,
            gameMode = gameMode
        }, response =>
        {
            var data = response.GetValue<JsonElement>(0);
            roomCode = data.GetProperty("roomCode").GetString();
            Debug.Log($"[Network] Room created: {roomCode}");
        });
    }

    public void JoinRoom(string code, string name)
    {
        socket.Emit("room:join", new
        {
            roomCode = code,
            playerName = name,
            password = (string)null
        }, response =>
        {
            bool success = response.GetValue<JsonElement>(0).GetProperty("success").GetBoolean();
            Debug.Log($"[Network] Join room: {(success ? "OK" : "Failed")}");
        });
    }

    public void StartGame()
    {
        socket.Emit("game:start", new { roleConfig = (object)null });
    }

    public void SendPosition(Vector3 position, Quaternion rotation)
    {
        socket.Emit("player:position", new
        {
            x = position.x,
            y = position.y,
            z = position.z,
            rotY = rotation.eulerAngles.y
        });
    }

    public void Submit3DAttack(string targetId)
    {
        socket.Emit("3d:attack", new { targetId });
    }

    public void Submit3DHide(string hideSpotId)
    {
        socket.Emit("3d:hide", new { hideSpotId });
    }

    public void SubmitNightAction(string action, string target, object ability)
    {
        socket.Emit("night:action", new { action, target, ability });
    }

    #endregion

    void OnDestroy()
    {
        socket?.DisconnectAsync();
    }
}
```

### 4.2 位置同步 (10Hz)

**Unity 端发送 (PlayerController3D.cs):**
```csharp
public class PlayerController3D : MonoBehaviour
{
    private float syncInterval = 0.1f; // 10Hz
    private float syncTimer = 0f;
    private Vector3 lastSentPosition;
    private float positionThreshold = 0.05f; // 5cm 变化才同步

    void Update()
    {
        HandleMovement();

        // 位置同步
        syncTimer += Time.deltaTime;
        if (syncTimer >= syncInterval)
        {
            syncTimer = 0f;
            Vector3 currentPos = transform.position;
            if (Vector3.Distance(currentPos, lastSentPosition) > positionThreshold)
            {
                NetworkManager.Instance.SendPosition(currentPos, transform.rotation);
                lastSentPosition = currentPos;
            }
        }
    }

    void HandleMovement()
    {
        float horizontal = Input.GetAxis("Horizontal");
        float vertical = Input.GetAxis("Vertical");

        Vector3 move = transform.right * horizontal + transform.forward * vertical;
        float speed = isSprinting ? sprintSpeed : (isCrouching ? crouchSpeed : walkSpeed);
        transform.position += move * speed * Time.deltaTime;
    }
}
```

```csharp
public class PlayerController3D : MonoBehaviour
{
    [Header("Movement")]
    public float walkSpeed = 3f;
    public float sprintSpeed = 6f;
    public float crouchSpeed = 1.5f;
    public float mouseSensitivity = 2f;

    [Header("Stamina")]
    public float maxStamina = 100f;
    public float staminaDrainRate = 20f;
    public float staminaRegenRate = 10f;
    private float currentStamina;

    [Header("Stealth")]
    public float stealthLevel = 0f;     // 0-100
    public float hideMoveSpeed = 0.5f;

    [Header("State")]
    public bool canMove = false;
    public bool isSprinting = false;
    public bool isCrouching = false;
    public bool isHidden = false;
    public string currentHideSpot;

    private CharacterController characterController;
    private Camera playerCamera;
    private Animator animator;

    void Start()
    {
        characterController = GetComponent<CharacterController>();
        playerCamera = GetComponentInChildren<Camera>();
        animator = GetComponent<Animator>();
        currentStamina = maxStamina;

        Cursor.lockState = CursorLockMode.Locked;
    }

    void Update()
    {
        if (!canMove) return;

        HandleMouseLook();
        HandleMovement();
        HandleStamina();
        UpdateAnimator();
    }

    void HandleMouseLook()
    {
        float mouseX = Input.GetAxis("Mouse X") * mouseSensitivity;
        float mouseY = Input.GetAxis("Mouse Y") * mouseSensitivity;

        transform.Rotate(Vector3.up, mouseX);
        playerCamera.transform.Rotate(Vector3.right, -mouseY);
    }

    void HandleMovement()
    {
        float h = Input.GetAxis("Horizontal");
        float v = Input.GetAxis("Vertical");

        // Shift 冲刺
        isSprinting = Input.GetKey(KeyCode.LeftShift) && currentStamina > 0 && !isCrouching;

        // Ctrl 蹲伏
        if (Input.GetKeyDown(KeyCode.LeftControl))
            isCrouching = !isCrouching;

        float speed = isSprinting ? sprintSpeed :
                      isCrouching ? crouchSpeed :
                      isHidden ? hideMoveSpeed :
                      walkSpeed;

        Vector3 move = transform.right * h + transform.forward * v;
        characterController.SimpleMove(move * speed);
    }

    void HandleStamina()
    {
        if (isSprinting)
        {
            currentStamina -= staminaDrainRate * Time.deltaTime;
            if (currentStamina <= 0)
                currentStamina = 0;
        }
        else
        {
            currentStamina += staminaRegenRate * Time.deltaTime;
            if (currentStamina > maxStamina)
                currentStamina = maxStamina;
        }
    }

    // 由 UI 按钮或按键触发
    public void TryHide(string hideSpotId)
    {
        if (isHidden) return;

        // 检查是否靠近藏匿点
        // ... 距离检查逻辑 ...

        isHidden = true;
        currentHideSpot = hideSpotId;
        stealthLevel += 25f;
        NetworkManager.Instance.Submit3DHide(hideSpotId);
    }

    public void Unhide()
    {
        isHidden = false;
        currentHideSpot = null;
        stealthLevel = Mathf.Max(0, stealthLevel - 25f);
    }

    // 蚀者攻击
    public void TryAttack(string targetId)
    {
        NetworkManager.Instance.Submit3DAttack(targetId);
    }

    void UpdateAnimator()
    {
        if (animator == null) return;
        float speed = new Vector3(
            characterController.velocity.x, 0,
            characterController.velocity.z
        ).magnitude;

        animator.SetFloat("Speed", speed / sprintSpeed);
        animator.SetBool("IsCrouching", isCrouching);
        animator.SetBool("IsSprinting", isSprinting);
        animator.SetBool("IsHiding", isHidden);
    }
}
```

```javascript
// 进入3D夜晚
_enter3DNight() {
    this.nightStep = 'FREE_ROAM';
    this._clearPhaseTimeout();
    this.timeLeft = 120;  // 120秒自由时间

    this.broadcastPhaseChange();

    // 通知客户端进入3D夜晚
    this._io.to(this.id).emit('game:3dNightStart', {
        timeLeft: this.timeLeft,
        nightStep: 'FREE_ROAM',
        players: this.players.filter(p => p.alive).map(p => p.toJSON()),
    });

    // 120秒后强制天亮
    this._phaseTimeout = setTimeout(() => {
        this._end3DNight();
    }, 120000);

    // AI蚀者自动狩猎
    for (const bot of this.players.filter(p => p.alive && p.isBot)) {
        if (bot.isWolf()) {
            bot._3dAction = 'HUNT';
        } else {
            bot._3dAction = Math.random() < 0.4 ? 'HIDE' : 'ROAM';
        }
    }
}

// 玩家提交3D攻击
submit3DAttack(corruptedId, targetId) {
    // 验证: 夜晚、3D模式、攻击者在世、目标在世、攻击者是蚀者
    const corrupted = this.getPlayer(corruptedId);
    const target = this.getPlayer(targetId);

    // 8秒冷却检查
    if (corrupted._lastAttackTime && Date.now() - corrupted._lastAttackTime < 8000) {
        return { success: false, reason: 'COOLDOWN' };
    }

    corrupted._lastAttackTime = Date.now();

    // 追猎者短铳反击
    if (target.role === 'FLAME_TRACKER' && target.blunderbussUsable) {
        corrupted.alive = false;
        target.blunderbussUsable = false;
        return { success: true, result: 'COUNTERED' };
    }

    // 逃脱概率: 30%基础 + 特质加成 + 藏匿加成
    let escapeChance = 0.30;
    if (target.characterId === 'ORIC') escapeChance += 0.15;
    if (target.characterId === 'SKADI') escapeChance += 0.10;
    if (target.isHidden) escapeChance += 0.25;

    if (Math.random() < escapeChance) {
        return { success: true, result: 'ESCAPED' };
    }

    // 击杀
    target.alive = false;
    target._killedBy = corruptedId;
    return { success: true, result: 'KILLED' };
}
```
